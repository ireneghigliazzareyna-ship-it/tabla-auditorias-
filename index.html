WITH mis_auditores AS ( SELECT 'araceli lara' AS auditor_nombre UNION ALL SELECT 'adonay ramirezmacias' UNION ALL SELECT 'andyalois mendez' UNION ALL SELECT 'saul sanchezcantu' UNION ALL SELECT 'rosa garciasilva' UNION ALL SELECT 'blanca martinezblanco' UNION ALL SELECT 'irene ghigliazzareyna' UNION ALL SELECT 'brenda liraflores' UNION ALL SELECT 'alejandra carrizalestorres' UNION ALL SELECT 'montserrat covarrubias' ), auditorias_unificadas AS ( -- 1. Tabla Maestra (TMWH MU / Procesos generales) SELECT CAST(FECHA_HORA AS STRING) AS FECHA_HORA, UPPER(TRIM(PROCESO)) AS PROCESO, LOWER(TRIM(REPLACE(AUDITOR, '.', ' '))) AS AUDITOR, CAST(ID AS STRING) AS ID, 0.0 AS MONTO, COALESCE( SAFE_CAST(FECHA_HORA AS TIMESTAMP), SAFE.PARSE_TIMESTAMP('%Y-%m-%d %H:%M:%E*S%Ez', CAST(FECHA_HORA AS STRING)) ) AS TS_FECHA FROM `meli-bi-data.SBOX_LPDESA.BD_MAESTRA_AUDITORIAS_PI_OUT` WHERE AUDITOR IS NOT NULL AND TRIM(AUDITOR) != '' AND PROCESO IS NOT NULL AND TRIM(PROCESO) != '' AND CAD = 'MXNL01' UNION ALL -- 2. Tabla Control 1P Ubicaciones SELECT CAST(FECHA_HORA AS STRING) AS FECHA_HORA, '1P UBICACIONES' AS PROCESO, LOWER(TRIM(REPLACE(AUDITOR, '.', ' '))) AS AUDITOR, CAST(ID AS STRING) AS ID, 0.0 AS MONTO, COALESCE( SAFE_CAST(FECHA_HORA AS TIMESTAMP), SAFE.PARSE_TIMESTAMP('%Y-%m-%d %H:%M:%E*S%Ez', CAST(FECHA_HORA AS STRING)) ) AS TS_FECHA FROM `meli-bi-data.SBOX_LPDESA.LP_PI_RESULTADOS_CONTROL1P` WHERE AUDITOR IS NOT NULL AND TRIM(AUDITOR) != '' AND CAD = 'MXNL01' UNION ALL -- 3. Tabla Control 1P Checkin SELECT CAST(FECHA_HORA AS STRING) AS FECHA_HORA, UPPER(TRIM(COALESCE(PROCESO, '1P CHECKIN'))) AS PROCESO, LOWER(TRIM(REPLACE(AUDITOR, '.', ' '))) AS AUDITOR, CAST(ID AS STRING) AS ID, 0.0 AS MONTO, COALESCE( SAFE_CAST(FECHA_HORA AS TIMESTAMP), SAFE.PARSE_TIMESTAMP('%Y-%m-%d %H:%M:%E*S%Ez', CAST(FECHA_HORA AS STRING)) ) AS TS_FECHA FROM `meli-bi-data.SBOX_LPDESA.LP_PI_RESULTADOS_CONTROL1P_CHECKIN` WHERE AUDITOR IS NOT NULL AND TRIM(AUDITOR) != '' AND CAD = 'MXNL01' UNION ALL -- 4. Retiros IN SELECT CAST(FECHA_HORA AS STRING) AS FECHA_HORA, UPPER(TRIM(COALESCE(PROCESO, 'RETIROS IN'))) AS PROCESO, LOWER(TRIM(REPLACE(AUDITOR, '.', ' '))) AS AUDITOR, CAST(ID AS STRING) AS ID, 0.0 AS MONTO, COALESCE( SAFE_CAST(FECHA_HORA AS TIMESTAMP), SAFE.PARSE_TIMESTAMP('%Y-%m-%d %H:%M:%E*S%Ez', CAST(FECHA_HORA AS STRING)) ) AS TS_FECHA FROM `meli-bi-data.SBOX_LPDESA.BD_MAESTRA_AUDITORIAS_PI_RETIROS_IN` WHERE AUDITOR IS NOT NULL AND TRIM(AUDITOR) != '' AND CAD = 'MXNL01' UNION ALL -- 5. CAZA LOST (LÓGICA EXACTA Y RIGUROSA: NO INFLA FOUNDS) SELECT CAST(L.DATA_RESULTADO_ORIGEN AS STRING) AS FECHA_HORA, CASE WHEN F.INVENTORY_ID IS NOT NULL THEN 'CAZA - FOUND' ELSE 'CAZA - DFL' END AS PROCESO, LOWER(TRIM(REPLACE( COALESCE( NULLIF(TRIM(L.RESPONSAVEL_LP_ORIGEN), ''), REP.NOME_DO_FUNCIONARIO ), '.', ' ' ))) AS AUDITOR, CAST( CASE WHEN L.INVENTORY_ID IS NULL OR TRIM(L.INVENTORY_ID) = '' THEN 'SIN_ID' ELSE UPPER(TRIM(L.INVENTORY_ID)) END AS STRING ) AS ID, CAST(COALESCE(L.PRECIO_USD, 0) AS FLOAT64) AS MONTO, SAFE_CAST(L.DATA_RESULTADO_ORIGEN AS TIMESTAMP) AS TS_FECHA FROM `meli-bi-data.WHOWNER.DM_LP_FBM_CAZA_LOST_TABLEAU` AS L -- Solo se marca FOUND si la tabla oficial de Founds de Caza Lost lo reconoce LEFT JOIN ( SELECT DISTINCT UPPER(TRIM(INVENTORY_ID)) AS INVENTORY_ID FROM `meli-bi-data.WHOWNER.DM_LP_FBM_CAZA_LOST_TABL_LP_FOUNDS` WHERE INVENTORY_ID IS NOT NULL AND TRIM(INVENTORY_ID) != '' ) AS F ON UPPER(TRIM(L.INVENTORY_ID)) = F.INVENTORY_ID AND UPPER(TRIM(L.INVENTORY_ID)) != '' -- Rescate directo de nombre de auditor si en Caza viene el ID Meli LEFT JOIN `meli-bi-data.SBOX_LPDESA.REPS_GR` AS REP ON TRIM(L.RESPONSAVEL_LP_ORIGEN) = CAST(REP.ID_MELI AS STRING) WHERE L.WAREHOUSE_ID = 'MXNL01' AND COALESCE( NULLIF(TRIM(L.RESPONSAVEL_LP_ORIGEN), ''), REP.NOME_DO_FUNCIONARIO ) IS NOT NULL UNION ALL -- 6. FOUNDS WMS DIRECTO (INVENTARIO Y MOVIMIENTOS PENDING/FOUND) SELECT CAST(mov.FBM_CREATED_DATE AS STRING) AS FECHA_HORA, 'FOUND WMS DIRECTO' AS PROCESO, LOWER(TRIM(REPLACE(rep.NOME_DO_FUNCIONARIO, '.', ' '))) AS AUDITOR, CAST(UPPER(TRIM(mov.INVENTORY_ID)) AS STRING) AS ID, 0.0 AS MONTO, SAFE_CAST(mov.FBM_CREATED_DATE AS TIMESTAMP) AS TS_FECHA FROM `meli-bi-data.WHOWNER.BT_FBM_STOCK_3_MOVEMENT` AS mov LEFT JOIN `meli-bi-data.SBOX_LPDESA.REPS_GR` AS rep ON mov.FBM_USER_ID = rep.ID_MELI WHERE mov.WAREHOUSE_ID = 'MXNL01' AND LOWER(mov.FBM_REASON_PROCESS) = 'found' ) -- FILTRADO FINAL Y LIMPIDO POR LISTA BLANCA DE TUS 10 AUDITORES SELECT u.FECHA_HORA, u.PROCESO, u.AUDITOR, u.ID, u.MONTO FROM auditorias_unificadas u INNER JOIN mis_auditores ma ON u.AUDITOR LIKE CONCAT('%', ma.auditor_nombre, '%') WHERE u.TS_FECHA >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 60 DAY) ORDER BY u.FECHA_HORA DESC;
